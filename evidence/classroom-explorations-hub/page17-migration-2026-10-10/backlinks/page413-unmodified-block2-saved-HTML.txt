            <ul>
              <li><strong>Cap:</strong> The top of the mushroom.</li>
              <li><strong>Gills:</strong> Thin sheets under the cap that release spores.</li>
              <li><strong>Stem:</strong> Supports the cap.</li>
              <li><strong>Mycelium:</strong> Thread-like network under the surface (the main body!).</li>

            <ol>

              <li>Pin →</li>

(function(){
  var page = document.getElementById('hrv-page');
  if(!page) return;

  var opener = page.querySelector('.hrv-opener');
  var skip = page.querySelector('.hrv-skip');

  function closeOpener(){
    if(opener){
      opener.style.display = 'none';
      opener.setAttribute('aria-hidden','true');
    }
  }

  if(skip){
    skip.addEventListener('click', closeOpener);
  }

  window.setTimeout(closeOpener, 2200);

  var frame = page.querySelector('#exploreVideo');
  var buttons = page.querySelectorAll('.vbtn');

  function toEmbed(url){
    try{
      var u = new URL(url);
      var id = '';
      if(u.hostname === 'youtu.be'){
        id = u.pathname.replace('/','');
      }else if(u.hostname.indexOf('youtube.com') !== -1 || u.hostname.indexOf('youtube-nocookie.com') !== -1){
        if(u.pathname === '/watch' && u.searchParams.get('v')){
          id = u.searchParams.get('v');
        }else if(u.pathname.indexOf('/embed/') === 0){
          id = u.pathname.split('/embed/')[1].split('/')[0];
        }
      }
      if(id){
        return 'https://www.youtube-nocookie.com/embed/' + id;
      }
      return url;
    }catch(e){
      return url;
    }
  }

  if(frame && frame.src){
    frame.src = toEmbed(frame.src) + '?rel=0';
  }

  buttons.forEach(function(btn){
    btn.addEventListener('click', function(){
      var raw = btn.getAttribute('data-src');
      var url = toEmbed(raw);
      if(url && frame){
        frame.src = url + (url.indexOf('?') > -1 ? '&' : '?') + 'rel=0';
        buttons.forEach(function(b){
          b.classList.remove('active');
        });
        btn.classList.add('active');
      }
    });
  });
})();
