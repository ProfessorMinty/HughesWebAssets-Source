<!-- ===================== Life Cycle Explorer ===================== -->
<section class="lifecycle reveal" role="region" aria-label="Butterfly Life Cycle">
  <h2>The Butterfly Life Cycle</h2>
  <div class="steps">
    <div class="step">
      <h3>1) Egg</h3>
      <p>Tiny eggs are laid on host plants (like cabbage!).</p>
      <p class="didyouknow"><strong>Did you know?</strong> Parents choose plants that will feed their babies.</p>
    </div>
    <div class="step">
      <h3>2) Caterpillar (Larva)</h3>
      <p>Chomp, chomp! Caterpillars eat and grow, shedding their skin in stages called instars.</p>
      <p class="didyouknow"><strong>Did you know?</strong> Those little black dots in our habitat are frass (poop) — a
        sign of healthy eaters!</p>
    </div>
    <div class="step">
      <h3>3) Chrysalis (Pupa)</h3>
      <p>Inside the chrysalis, the body is re-arranged into a butterfly. It’s quiet on the outside, busy on the inside.
      </p>
      <p class="didyouknow"><strong>Did you know?</strong> Phil and Green Bean were our pioneers into this stage!</p>
    </div>
    <div class="step">
      <h3>4) Butterfly</h3>
      <p>Wings pump up and dry. Then—lift off! We’ll release them outdoors where they belong.</p>
    </div>
  </div>
</section>

  <h2>Words We’re Using</h2>
  <ul class="vocab-list">
    <li><strong>Metamorphosis:</strong> changing shape from caterpillar to butterfly.</li>
    <li><strong>Chrysalis:</strong> the protective case where change happens.</li>
    <li><strong>Instar:</strong> a growth stage between molts.</li>
    <li><strong>Camouflage:</strong> blending in to stay safe.</li>

<script>
  /* Scroll reveal */
  const io = new IntersectionObserver((entries) => {
    entries.forEach(e => {
      if (e.isIntersecting) { e.target.classList.add('is-visible'); io.unobserve(e.target); }
    });
  }, { threshold: .12 });
  document.querySelectorAll('.reveal').forEach(el => io.observe(el));

  /* (Optional) Tap-to-pause parade on mobile for fun */
  const parade = document.querySelector('.parade');
  if (parade) {
    parade.addEventListener('click', () => {
      parade.style.animationPlayState = parade.style.animationPlayState === 'paused' ? 'running' : 'paused';
      document.querySelectorAll('.crawler').forEach(c => {
        c.style.animationPlayState = c.style.animationPlayState === 'paused' ? 'running' : 'paused';
      });
    });
  }
</script>
